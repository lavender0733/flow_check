```python
"""測試共用工具。

所有測試都在 tempfile 目錄中執行，絕不觸碰真實的 data/。
"""

import sys
import shutil
import tempfile
import unittest
from pathlib import Path

# 專案根目錄（tests/ 的上一層）
ROOT = Path(__file__).resolve().parents[1]


def add_source_paths() -> None:
    """把原始碼目錄加進 sys.path，讓測試能直接 import 各模組。

    冪等：重複呼叫不會重複插入。
    階段 B（Task 11）把程式改成套件後，這裡只保留 program/。
    """
    for rel in ("program/core", "program/gpss", "program"):
        p = str(ROOT / rel)
        if p not in sys.path:
            sys.path.insert(0, p)


# 不要在 Google Drive 同步夾裡寫 .pyc（與正式程式一致）
sys.dont_write_bytecode = True
add_source_paths()


class TempDataCase(unittest.TestCase):
    """提供一個乾淨的暫存目錄 self.tmp，測試結束自動刪除。"""

    def setUp(self):
        self.tmp = Path(tempfile.mkdtemp(prefix="flowcheck-test-"))

    def tearDown(self):
        shutil.rmtree(self.tmp, ignore_errors=True)

    def write_json(self, name: str, obj) -> Path:
        """在暫存目錄寫一個 JSON 檔，回傳路徑。"""
        import json
        p = self.tmp / name
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(json.dumps(obj, ensure_ascii=False), encoding="utf-8")
        return p
```

```python
    """回傳一個 datetime 子類別，其 now() 固定在指定時刻。

    quota.py 是 `from datetime import datetime`，所以要換掉的是 quota 模組
```

```python
"""JSON 讀寫與檔案簽章快取。

零專案內相依的葉模組：不 import 專案內任何其他模組，任何地方都能安全使用。

取代原本散在各處的重複實作：
- 原子寫入 6 處（traffic_monitor / storage / monthly_progress / output_store ×2 / yearly_server）
- 容錯讀 JSON 18 處
- 檔案簽章 3 處（格式原本不一致，見下方 file_sig 的說明）
"""

import json
import time
from pathlib import Path


def read_json(path, default=None):
    """讀 JSON。讀不到、壞檔、空檔、路徑是目錄——一律回 default，絕不拋例外。

    呼叫端遍布全專案（多半在 Google Drive 同步夾上讀檔，同步中暫時讀不到很常見），
    統一在這裡吞掉，呼叫端就不必各自包 try/except。
    """
    try:
        return json.loads(Path(path).read_text(encoding="utf-8"))
    except (OSError, ValueError, TypeError):
        return default


def write_json_atomic(path, obj, retries: int = 4) -> None:
    """先寫 <檔名>.tmp、寫完才 replace() 一次換上。

    直接覆寫在 Google Drive 同步夾上特別危險：同步層可能在寫到一半時讀走，
    留下半截 JSON。原子替換讓讀取端永遠看到完整的舊檔或完整的新檔。

    同步磁碟短暫忙碌時做有限重試；全部失敗才拋出最後一個例外
    （呼叫端要知道沒寫成功，不可默默吞掉）。
    """
    path = Path(path)
    # 先序列化：資料本身有問題時直接拋，不要先動到磁碟上的舊檔
    data = json.dumps(obj, ensure_ascii=False, indent=2)
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(path.name + ".tmp")
    last_exc = None
    for attempt in range(max(1, retries)):
        try:
            tmp.write_text(data, encoding="utf-8")
            tmp.replace(path)
            return
        except OSError as e:
            last_exc = e
            time.sleep(0.1 * (attempt + 1))
    try:
        tmp.unlink()
    except OSError:
        pass
    raise last_exc


def file_sig(path, version=None) -> str:
    """檔案的輕量變動簽章（大小 + 修改時間奈秒）；讀不到回空字串。

    格式為 "<size>:<mtime_ns>"，帶 version 時附加 ":v<version>"。
    帶 version 的形式與 yearly_server 原本寫進 data/年度統計快取.json 的格式
    逐字相同，因此既有的磁碟快取不會失效。

    用 mtime_ns 而非截斷到秒的 mtime：同一秒內的修改也抓得到。
    """
    try:
        st = Path(path).stat()
    except (OSError, TypeError):
        return ""
    base = f"{st.st_size}:{st.st_mtime_ns}"
    return f"{base}:v{version}" if version is not None else base


class SigCache:
    """以檔案簽章為鍵的解析結果快取。

    用途：Excel / 匯出檔的解析很貴，但檔案只在使用者換新檔時才變。
    簽章沒變就回上次的解析結果。

    version 用於「解析結果的欄位改版」：欄位一改，舊快取的形狀就對不上，
    換一個 version 即可讓所有舊項目自動失效，不必手動刪快取檔。
    """

    def __init__(self, version=None):
        self.version = version
        self._items = {}     # str(path) -> (sig, data)

    def get(self, path, parse):
        """簽章命中就回快取；否則呼叫 parse(path) 重新解析。

        回傳 (data, hit)。hit 為 True 代表這次沒有重新解析。
        檔案讀不到時（簽章為空）一律視為未命中，不快取。
        """
        key = str(path)
        sig = file_sig(path, self.version)
        if sig:
            cached = self._items.get(key)
            if cached and cached[0] == sig:
                return cached[1], True
        data = parse(path)
        if sig:
            self._items[key] = (sig, data)
        return data, False

    def clear(self) -> None:
        self._items.clear()
```

```python
"""機器身分與驗證碼遮罩。

零專案內相依的葉模組。取代原本重複的四份 machine_id() 與三份 mask_code()。
"""

import re
import socket


def machine_id() -> str:
    """本機代號：主機名稱過濾成檔名安全字元。

    這個值會嵌進檔名（state_<機器>.json、run_log_<機器>.jsonl 等），
    讓 Google Drive 同步時各機不互相覆蓋。身分只能取自主機名稱——
    同步資料夾裡的檔案兩台看到的都一樣。
    """
    raw = socket.gethostname() or "PC"
    mid = re.sub(r"[^A-Za-z0-9]+", "-", raw).strip("-")
    return mid or "PC"


def mask_code(code) -> str:
    """只顯示前 4 碼，其餘以 * 取代（保留長度）。

    統一自三份實作；採用其中「有 strip」的版本——額度卡與月進度卡
    兩張都用它，顯示必須對齊。
    """
    code = str(code or "").strip()
    return code if len(code) <= 4 else code[:4] + "*" * (len(code) - 4)
```

```python
def call(handler_cls, method, path, body=None):
    """對 handler 送一個請求，回傳 (status_code, headers: dict, body: bytes)。"""
    payload = b"" if body is None else json.dumps(body).encode("utf-8")
    raw = (f"{method} {path} HTTP/1.1\r\n"
           f"Host: 127.0.0.1\r\n"
           f"Content-Length: {len(payload)}\r\n"
           f"\r\n").encode("utf-8") + payload

    req = FakeRequest(raw)
    handler_cls(req, ("127.0.0.1", 12345), None)
    out = req.sent.getvalue()

    head, _, rest = out.partition(b"\r\n\r\n")
    lines = head.decode("latin-1").split("\r\n")
    code = int(lines[0].split()[1])
    headers = {}
    for line in lines[1:]:
        if ":" in line:
            k, _, v = line.partition(":")
            headers[k.strip().lower()] = v.strip()
    return code, headers, rest
```

```python
"""本機 HTTP 伺服器底座。

兩支伺服器（流量監控 8765、年度統計 8766）原本各抄一套一模一樣的管線：
送 JSON、服務靜態檔、路徑穿越防護、快取標頭、靜音 log、吞掉瀏覽器斷線。
全部收斂到這裡；再開第三個小工具時只要寫一張路由表。

路由是 (method, path) -> handler 的精確比對，不是 startswith。舊版用
startswith 有兩個問題：/api/monthly/trend 必須排在 /api/monthly 之前
否則被靜默吃掉，而 /api/statusXYZ 這種亂打的路徑會被誤收。
"""

import json
import traceback
import webbrowser
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from urllib.parse import urlparse, parse_qs, unquote

# 只服務這些副檔名。網頁用得到的就這三種；其他一律不給，
# 免得有人把含 userCode 的 .json 放進 web/ 就被整份端出去。
STATIC_TYPES = {
    ".css": "text/css",
    ".js": "application/javascript",
    ".html": "text/html",
}

NO_STORE = "no-store, must-revalidate"


class Request:
    """一個已解析好的請求。handler 只需要看這四個欄位。"""

    __slots__ = ("method", "path", "query", "body")

    def __init__(self, method, path, query, body):
        self.method = method
        self.path = path      # 已去除 query string
        self.query = query    # {str: str}，重複參數取第一個
        self.body = body      # POST 的 JSON；非 JSON 或空 body 為 {}


class Raw:
    """要送出非 JSON 內容時，handler 回傳這個。"""

    __slots__ = ("body", "content_type", "code")

    def __init__(self, body: bytes, content_type: str, code: int = 200):
        self.body = body
        self.content_type = content_type
        self.code = code


def build_handler(routes, static_dir=None, index_renderer=None,
                  on_error=None, quiet=True):
    """造一個 BaseHTTPRequestHandler 子類別。

    routes:         {(method, path): handler}，handler(req) 回傳可 JSON 序列化
                    的物件，或 Raw。
    static_dir:     靜態檔根目錄；None 表示不服務靜態檔。
    index_renderer: 無參數可呼叫物件，回傳 "/" 的 HTML bytes。
                    None 時 "/" 退回讀 static_dir/index.html。
    on_error:       on_error(where, exc)，處理端點例外時呼叫（記錄用）。
    quiet:          True 時不印每個請求的存取記錄。
    """
    static_root = Path(static_dir).resolve() if static_dir else None

    class Handler(BaseHTTPRequestHandler):
        # 讓瀏覽器的 keep-alive 正常運作（與舊版一致）
        protocol_version = "HTTP/1.0"

        # ── 送出 ────────────────────────────────────────────
        def _send_bytes(self, body: bytes, content_type: str, code: int = 200):
            self.send_response(code)
            self.send_header("Content-Type", content_type)
            self.send_header("Content-Length", str(len(body)))
            # 不讓瀏覽器快取：否則改了網頁/JS，重整仍可能跑到舊版。
            self.send_header("Cache-Control", NO_STORE)
            self.end_headers()
            self.wfile.write(body)

        def _send_json(self, obj, code: int = 200):
            body = json.dumps(obj, ensure_ascii=False).encode("utf-8")
            self._send_bytes(body, "application/json; charset=utf-8", code)

        def _send_status(self, code: int):
            self.send_response(code)
            self.send_header("Content-Length", "0")
            self.end_headers()

        # ── 解析 ────────────────────────────────────────────
        def _build_request(self, method):
            parsed = urlparse(self.path)
            query = {k: v[0] for k, v in parse_qs(parsed.query).items() if v}
            body = {}
            if method == "POST":
                try:
                    length = int(self.headers.get("Content-Length", 0) or 0)
                except ValueError:
                    length = 0
                if length > 0:
                    try:
                        body = json.loads(self.rfile.read(length).decode("utf-8"))
                    except (ValueError, UnicodeDecodeError):
                        body = {}
                    if not isinstance(body, dict):
                        body = {}
            return Request(method, unquote(parsed.path), query, body)

        # ── 分派 ────────────────────────────────────────────
        def _dispatch(self, method):
            req = self._build_request(method)
            handler = routes.get((method, req.path))
            if handler is not None:
                try:
                    result = handler(req)
                except Exception as exc:
                    if on_error:
                        on_error(req.path, exc)
                    else:
                        traceback.print_exc()
                    self._send_json({"ok": False, "message": str(exc)}, code=500)
                    return
                if isinstance(result, Raw):
                    self._send_bytes(result.body, result.content_type, result.code)
                else:
                    self._send_json(result)
                return
            if method == "GET":
                self._serve_static(req.path)
                return
            self._send_status(404)

        def do_GET(self):
            self._dispatch("GET")

        def do_POST(self):
            self._dispatch("POST")

        # ── 靜態檔 ──────────────────────────────────────────
        def _serve_static(self, path):
            if path in ("/", "/index.html") and index_renderer is not None:
                try:
                    body = index_renderer()
                except Exception as exc:
                    if on_error:
                        on_error("index", exc)
                    self._send_json({"ok": False, "message": str(exc)}, code=500)
                    return
                self._send_bytes(body, "text/html; charset=utf-8")
                return

            if static_root is None:
                self._send_status(404)
                return

            rel = "index.html" if path in ("/", "/index.html") else path.lstrip("/")
            target = (static_root / rel).resolve()
            # 路徑穿越防護：解析後必須仍在 static_root 底下。
            try:
                target.relative_to(static_root)
            except ValueError:
                self._send_status(403)
                return
            ctype = STATIC_TYPES.get(target.suffix)
            if ctype is None:
                self._send_status(403)
                return
            try:
                body = target.read_bytes()
            except (OSError, ValueError):
                self._send_status(404)
                return
            self._send_bytes(body, ctype + "; charset=utf-8")

        # ── 雜項 ────────────────────────────────────────────
        def handle(self):
            # 瀏覽器切分頁/重整/關頁時會中途斷線（WinError 10053/10054）。
            # 對這種每幾秒輪詢一次的頁面很常見、也無害 —— 吞掉，別讓主控台
            # 噴一整串 traceback。
            try:
                super().handle()
            except (ConnectionAbortedError, ConnectionResetError, BrokenPipeError):
                pass
            except Exception as exc:
                if on_error:
                    on_error("http", exc)

        def log_message(self, fmt, *args):
            if not quiet:
                print(f"[web] {self.address_string()} - {fmt % args}")

    return Handler


def serve(port, routes, static_dir=None, index_renderer=None, on_error=None,
          quiet=True, banner=None, open_browser=True) -> None:
    """綁定連接埠並開始服務，直到 Ctrl+C。

    先綁定再開瀏覽器：連接埠一綁好就開始 listen，此刻開瀏覽器第一個請求
    一定連得上 —— 不會出現「無法連上 127.0.0.1」、要重整才載得出來。
    """
    handler_cls = build_handler(routes, static_dir=static_dir,
                                index_renderer=index_renderer,
                                on_error=on_error, quiet=quiet)
    httpd = ThreadingHTTPServer(("127.0.0.1", port), handler_cls)
    url = f"http://127.0.0.1:{port}"
    if banner:
        print(banner)
    if open_browser:
        webbrowser.open(url)
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\n停止。")
        httpd.shutdown()
    finally:
        httpd.server_close()
```

```python
class HttpError(Exception):
    """端點要回非 200 狀態碼時拋這個。

    payload 是要送出的 dict；傳字串時自動包成 {"ok": False, "message": ...}
    （現行絕大多數錯誤回應就是這個形狀）。
    """

    def __init__(self, code: int, payload):
        super().__init__(payload if isinstance(payload, str) else str(payload))
        self.code = code
        self.payload = ({"ok": False, "message": payload}
                        if isinstance(payload, str) else payload)
```

```python
# index.html 的三個組裝標記。partials 與 script 都改為掃目錄產生 ——
# 新增一個頁籤只要丟 web/js/<名稱>.js + web/partials/<名稱>.html，
# index.html 一字都不必改。
_MARK_PARTIALS = "<!--#include-partials-->"
_MARK_SCRIPTS = "<!--#include-scripts-->"
_MARK_VERSION = "<!--#version-->"

# 載入順序只有兩條硬性規則：util.js 最先（它定義 registerTab），
# main.js 最後（它是啟動者）。中間照字母序 —— 已確認那些檔的跨檔引用
# 全在函式內（執行期），頂層只有 main.js 真的執行啟動碼。
_JS_FIRST = "util.js"
_JS_LAST = "main.js"


def _ordered_scripts() -> list:
    """web/js/*.js 依載入順序排好的檔名清單。"""
    names = sorted(p.name for p in (WEB_DIR / "js").glob("*.js"))
    first = [n for n in names if n == _JS_FIRST]
    last = [n for n in names if n == _JS_LAST]
    middle = [n for n in names if n not in (_JS_FIRST, _JS_LAST)]
    return first + middle + last


def _partials_html() -> str:
    """web/partials/*.html 的內容串接（面板同時只顯示一個，DOM 順序無關）。"""
    d = WEB_DIR / "partials"
    if not d.is_dir():
        return ""
    return "\n".join(p.read_text(encoding="utf-8")
                     for p in sorted(d.glob("*.html")))


def render_index(version: str) -> bytes:
    """組裝 index.html：注入 partials、產生 script 標籤、換上版號。

    每次請求即時重讀（改了 partial 或 js，重整即生效，不必重啟）。
    """
    html = (WEB_DIR / "index.html").read_text(encoding="utf-8")
    scripts = "\n".join(f'<script src="js/{n}"></script>'
                        for n in _ordered_scripts())
    html = html.replace(_MARK_PARTIALS, _partials_html())
    html = html.replace(_MARK_SCRIPTS, scripts)
    html = html.replace(_MARK_VERSION, version)
    return html.encode("utf-8")
```
